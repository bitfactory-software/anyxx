121template <std::meta::info Spec, std::meta::info AnyVal, typename V, typename R, typename VoidSelf, typename... Args>
122consteval invoke_function_t<impl_return_type<R, V>, VoidSelf, Args...> default_impl() {
123 using return_t = impl_return_type<R, V>;
124 using self_t = self_const_correct_t<V, VoidSelf>;
125 if constexpr(is_class_type(^^V)) {
126 if constexpr(is_variant<V>) {
127 return invoke_variant_void<Spec, AnyVal, V, return_t, VoidSelf,Args...>;
128 } else {
129 if constexpr(constexpr auto candidate = find_candidate_in_target<Spec, V, return_t, VoidSelf, Args...>(); candidate != std::meta::info{}) {
130 return invoke_member<candidate, self_t, return_t, VoidSelf, Args...>;
131 }
132 }
133 }
134 if constexpr(meta::is_op_parentheses_spec(Spec)) {
135 if constexpr(std::is_invocable_r_v<return_t, self_t, Args...>) {
136 return invoke_op_parentheses<self_t, return_t, VoidSelf, Args...>;
137 } else {
138 std::string msg{display_string_of(^^V)};
139 msg += " has no implementation for " + std::string{display_string_of(Spec)};
140 throw std::meta::exception(msg, ^^V);
141 }
142 } else {
143 if constexpr(!is_operator_function(Spec)) {
144 std::string msg{ display_string_of(^^V) };
145 msg += " has no member function " + std::string{display_string_of(Spec)};
146 throw std::meta::exception(msg, ^^V);
147 } else {
148#define __RETURN_OP(op_name) \
149 if constexpr(meta::is_op_spec(Spec, std::meta::op_##op_name)) { \
150 return invoke_op_##op_name<self_t, return_t, VoidSelf, Args...>; \
151 }
152
153 __RETURN_OP(plus_plus)
154 else __RETURN_OP(minus_minus)
155 else __RETURN_OP(star)
156 else __RETURN_OP(arrow)
157 else __RETURN_OP(square_brackets)
158 else __RETURN_OP(plus)
159 else __RETURN_OP(minus)
160 else __RETURN_OP(equals_equals)
161 else __RETURN_OP(exclamation_equals)
162 else __RETURN_OP(less)
163 else __RETURN_OP(less_equals)
164 else __RETURN_OP(greater)
165 else __RETURN_OP(greater_equals)
166 else __RETURN_OP(plus_equals)
167 else __RETURN_OP(minus_equals)
168
169#undef __RETURN_OP
170 else {
171 std::string msg{ display_string_of(Spec) };
172 msg += " not yet implemeted in anyxx " + std::string{ display_string_of(Spec) };
173 throw std::meta::exception(msg, Spec);
174 }
175 }
176 }

179template <bool default_, std::meta::info m, std::meta::info dyn_self_val, std::meta::info dyn_self_cref, std::meta::info dyn_self_mutref,
180 typename V, typename R, typename VoidSelf, typename... Args>
181constexpr [:translate_v_table_return_type<dyn_self_val>(^^R):] vfimpl(VoidSelf void_self, translate_v_table_fptr_param_type_t<dyn_self_cref, dyn_self_mutref, Args>... args) {
182 using return_t = [:translate_v_table_return_type<dyn_self_val>(^^R):];
183 if constexpr (default_) {
184 constexpr auto fptr = default_impl<m, dyn_self_val, V, R, VoidSelf, impl_fptr_param_t<V, Args>...>();
185 if constexpr(std::same_as<return_t, void>) {
186 fptr(void_self, forward_v_table_fptr_param<V, Args>(args)...);
187 return;
188 } else {
189 return fptr(void_self, forward_v_table_fptr_param<V, Args>(args)...);
190 }
191 } else {
192 using VSelf = self_const_correct_t<V, VoidSelf>;
193 if constexpr(std::same_as<return_t, void>) {
194 [:m:](*static_cast<VSelf*>(void_self), forward_v_table_fptr_param<V, Args>(args)...);
195 return;
196 } else {
197 return [:m:](*static_cast<VSelf*>(void_self), forward_v_table_fptr_param<V, Args>(args)...);
198 }
199 }

202consteval std::meta::info make_vfimpl(std::meta::info concrete_type, std::meta::info implementing_function, std::meta::info dyn_self_val, std::meta::info dyn_self_cref, std::meta::info dyn_self_mutref, std::meta::info interface_function) {
203 std::vector<std::meta::info> types;
204 types.push_back(std::meta::reflect_constant(is_defaulted_function_spec(implementing_function)));
205 types.push_back(reflect_constant(implementing_function));
206 types.push_back(std::meta::reflect_constant(dyn_self_val));
207 types.push_back(std::meta::reflect_constant(dyn_self_cref));
208 types.push_back(std::meta::reflect_constant(dyn_self_mutref));
209 types.push_back(concrete_type);
210 types.push_back(return_type_of(interface_function));
211 add_v_table_fptr_this_param_type(interface_function, types);
212 for (auto p : define_static_array(parameters_of(interface_function))) {
213 types.push_back(make_v_table_fptr_param_type(types.size() == 7u, p));
214 }
215 return substitute(^^vfimpl, types);
216}

222template <typename VTable, std::meta::info dyn_self_val, std::meta::info dyn_self_cref, std::meta::info dyn_self_mutref, typename Concrete, std::meta::info FunctionPointers>
223void set_v_table_members(VTable* v_table) {
224 template for(constexpr auto v_table_spec : define_static_array(get_interface_specs(^^typename VTable::trait_declaration_t))){
225 constexpr auto v_table_entry = anyxx::meta::get_data_member_by_id(FunctionPointers, v_table_name_of(v_table_spec));
226 if constexpr(is_v_table_data(v_table_spec)) {
227 v_table->[:v_table_entry:] = [:v_table_spec.member:]::template init<Concrete>(v_table);
228 } else {
229 constexpr auto trait_template = template_of(v_table_spec.declaration_trait);
230 constexpr auto args = std::define_static_array(template_arguments_of(v_table_spec.declaration_trait) | std::views::drop(1));
231 constexpr auto implemenation_member = find_function_impl(v_table_spec.member, trait_template, ^^Concrete, args);
232 v_table->[:v_table_entry:] = [:make_vfimpl(^^Concrete, implemenation_member, dyn_self_val, dyn_self_cref, dyn_self_mutref, v_table_spec.member):];
233 }
