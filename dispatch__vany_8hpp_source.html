10template <typename DynamicDispatch, auto StaticDispatch>
11class dispatch_vany {
12 DynamicDispatch dynamic_dispatch_;
13 constexpr static const std::size_t dimension_count =
14 DynamicDispatch::dimension_count;
15
16 template <typename Vany1, typename... Args>
17 auto invoke1(Vany1&& vany, Args&&... args) const {
18 return std::visit(
19 [&]<typename TypedArg>([[maybe_unused]] TypedArg&& arg) {
20 return overloads{
21 StaticDispatch,
22 [&]<is_any Any, typename... Vargs>(Any&& any_, Vargs&&... vargs) {
23 return dynamic_dispatch_(std::forward<Any>(any_),
24 std::forward<Vargs>(vargs)...);
25 }}(std::forward<TypedArg>(arg), std::forward<Args>(args)...);
26 },
27 get_proxy_value(std::forward<Vany1>(vany)));
28 }
29
30 template <typename Vany1, typename Vany2, typename... Args>
31 auto invoke2(Vany1&& vany1, Vany2&& vany2, Args&&... args) const {
32 using vany1_t = std::decay_t<Vany1>;
33 using vany2_t = std::decay_t<Vany2>;
34 using cv1_t = anyxx::vany_type_trait<vany1_t>::concrete_variant;
35 using any_v1 = anyxx::vany_type_trait<vany1_t>::any_in_variant;
36 using cv2_t = anyxx::vany_type_trait<vany2_t>::concrete_variant;
37 using any_v2 = anyxx::vany_type_trait<vany2_t>::any_in_variant;
38
39 auto dispatch_combined = [&]<typename DA1, typename DA2>(DA1&& da1,
40 DA2&& da2) {
41 auto dyn_case1 = [&]<is_any A1, is_any A2, typename... VAs>(
42 A1&& a1, A2&& a2, VAs&&... vas) {
43 return dynamic_dispatch_(std::forward<A1>(a1), std::forward<A2>(a2),
44 std::forward<VAs>(vas)...);
45 };
46 auto dyn_case2 = [&]<is_any A1, typename A2, typename... VAs>(
47 A1&& a1, A2 a2, VAs&&... vas)
48 requires std::constructible_from<cv2_t, A2>
49 {
50 return dynamic_dispatch_(std::forward<A1>(a1),
51 any_v2{std::in_place_type<cv2_t>, cv2_t{std::move(a2)}},
52 std::forward<VAs>(vas)...);
53 };
54 auto dyn_case3 = [&]<typename A1, is_any A2, typename... VAs>(
55 A1 a1, A2&& a2, VAs&&... vas)
56 requires std::constructible_from<cv1_t, A1>
57 {
58 return dynamic_dispatch_(any_v1{std::in_place_type<cv1_t>, cv1_t{std::move(a1)}},
59 std::forward<A2>(a2),
60 std::forward<VAs>(vas)...);
61 };
62 auto dyn_case4 = [&]<typename A1, typename A2, typename... VAs>(
63 A1 a1, A2 a2, VAs&&... vas)
64 requires(std::constructible_from<cv1_t, A1> &&
65 std::constructible_from<cv2_t, A2>)
66 {
67 return dynamic_dispatch_(any_v1{std::in_place_type<cv1_t>, cv1_t{std::move(a1)}},
68 any_v2{std::in_place_type<cv2_t>, cv2_t{std::move(a2)}},
69 std::forward<VAs>(vas)...);
70 };
71 return overloads{StaticDispatch, dyn_case1, dyn_case2, dyn_case3,
72 dyn_case4}(std::forward<DA1>(da1),
73 std::forward<DA2>(da2),
74 std::forward<Args>(args)...);
75 };
76
77 return std::visit(dispatch_combined,
78 get_proxy_value(std::forward<Vany1>(vany1)),
79 get_proxy_value(std::forward<Vany2>(vany2)));
80 }
81
82 public:
83 template <typename... Classes>
84 auto define(auto f) {
85 return dynamic_dispatch_.template define<Classes...>(f);
86 }
87
88 template <typename... Args>
89 auto operator()(Args&&... args) const {
90 if constexpr (dimension_count == 1) {
91 return invoke1(std::forward<Args>(args)...);
92 } else {
93 if constexpr (dimension_count == 2) {
94 return invoke2(std::forward<Args>(args)...);
95 } else {
96 static_assert(dimension_count <= 2,
97 "dispatch_vany only supports one and two dimensions");
98 }
99 }
100 }
101};
