349 template <typename Matrix>
350 bool has_conflicts(dispatch_path path, Matrix const& matrix) {
351 auto top = path.front();
352 if constexpr (std::same_as<Matrix, std::vector<erased_function_t>>) {
353 if (matrix.size() <= top || !matrix[top]) {
354 return true;
355 } else {
356 return false;
357 }
358 } else {
359 path.pop_front();
360 if (top == 0) {
361 for (auto i : std::views::iota(1u, matrix.size())) {
362 if (i == path.front()) continue;
363 if (matrix[i].empty()) continue;
364 if (!matrix[i][0]) continue;
365 if (matrix[i].size() < path.front()) {
366 return true;
367 }
368 if (has_conflicts(path, matrix[i])) {
369 return true;
370 }
371 }
372 } else {
373 if (matrix.size() > top) return has_conflicts(path, matrix[top]);
374 }
375 return false;
376 }
377 }

401 template <typename DispatchMatrix, typename ArgsTuple,
402 typename... ActualArgs>
403 std::optional<R> invoke(DispatchMatrix const& target,
404 ArgsTuple&& dispatch_args_tuple, Any const& any_,
405 ActualArgs&&... actual_args) const {
406 auto dispatch_table = get_v_table(any_)->dispatch_table;
407 auto dispatch_dim = get_multi_dispatch_index_at(dispatch_table, index_);
408 if (dispatch_dim && target.size() > dispatch_dim)
409 if (auto found =
410 next_t::invoke(target[dispatch_dim],
411 std::forward<ArgsTuple>(dispatch_args_tuple),
412 std::forward<ActualArgs>(actual_args)...))
413 return found;
414
415 if (target.size())
416 return next_t::invoke(
417 target[0],
418 std::tuple_cat(get_tuple_head<Dimension>(dispatch_args_tuple),
419 std::make_tuple(&any_),
420 get_tuple_tail<Dimension + 1>(dispatch_args_tuple)),
421 std::forward<ActualArgs>(actual_args)...);
422 return {};
423 }
