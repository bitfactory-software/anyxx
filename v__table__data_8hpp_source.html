164template <typename Concrete>
165mutable_void invoke_move_constructor([[maybe_unused]] mutable_void placement,
166 [[maybe_unused]] mutable_void from) {
167 static_assert(std::is_move_constructible_v<Concrete>);
168 auto typed_placement = static_cast<Concrete*>(placement);
169 auto typed_from = static_cast<Concrete*>(from);
170 [[maybe_unused]] auto constructed =
171 std::construct_at<Concrete>(typed_placement, std::move(*typed_from));
172 assert(placement == constructed);
173 std::destroy_at(typed_from);
174 return placement;
175}
