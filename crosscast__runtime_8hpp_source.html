40template <is_any ToAny, is_dynamic_castable_v_table FromVTable>
41 requires is_dynamic_castable_v_table<typename ToAny::v_table_t>
42auto query_v_table(FromVTable* from)
43 -> std::expected<typename ToAny::v_table_t*, anyxx::cast_error> {
44 using v_table_t = typename ToAny::v_table_t;
45 if (from->is_derived_from_(typeid(v_table_t)))
46 return reinterpret_cast<v_table_t*>(from);
47 if constexpr (is_meta_data_v_table<FromVTable>) {
48 if (auto meta_data = from->meta_data_; meta_data) {
49 return meta_data->template get_v_table<v_table_t>();
50 }
51 }
52 return std::unexpected(
53 anyxx::cast_error{typeid(v_table_t), *from->type_info_});
54}

59 struct i_table_entry {
60 void* v_table_ = nullptr;

64 i_table_entry(VTable* v_table)
65 : v_table_(v_table), type_index_(typeid(*v_table)) {}

68 std::vector<i_table_entry> i_table_;
69
70 public:
71 template <typename CLASS>
72 explicit constexpr meta_data(std::in_place_type_t<CLASS>)
73 : type_info_(typeid(CLASS)) {}
74
75 constexpr const std::type_info& type() const { return type_info_; }
76
77 auto& get_i_table() { return i_table_; }
78 auto& get_i_table() const { return i_table_; }

80 template <typename VTable>
81 VTable* find_v_table() const {
82 if (auto found =
83 std::ranges::find(get_i_table(), std::type_index(typeid(VTable)),
84 &i_table_entry::type_index_);
85 found != get_i_table().end())
86 return static_cast<VTable*>(found->v_table_);
87 return nullptr;
88 }

90 template <typename VTable>
91 std::expected<VTable*, cast_error> get_v_table() const {
92 if (auto v_table = find_v_table<VTable>(); v_table) return v_table;
93 return std::unexpected(cast_error{.to = typeid(VTable), .from = type()});
94 }
95 template <typename VTable>
96 auto register_v_table(VTable* v_table) {
97 assert(v_table->meta_data_ == this);
98 if (!find_v_table<VTable>()) i_table_.push_back({v_table});
99 return v_table;
100 }
