Introduction to Internal and External Polymorphism:
using namespace anyxx;
template <typename Self>
struct Shape {
double area() const;
};
using any_shape = any_<shared, Shape>;
using any_shapes = std::vector<any_shape>;
void print_area_dyn(any_shapes const& shapes) {
std::println("Area: {}", std::ranges::fold_left(
shapes, 0.0, [](auto sum, any_shape const& s) {
return sum + s.area();
}));
}
template <typename... Args>
void print_area_static(trait_as<Args, Shape> const&... shapes) {
std::println("Area: {}", (shapes.area() + ...));
}
struct Circle {
double r = 0.0;
};
template <>
struct Shape<Circle> {
static double area(const Circle& c) { return M_PI * c.r * c.r; }
};
struct Rect {
double w, h;
};
template <>
struct Shape<Rect> {
static double area(const Rect& r) { return r.w * r.h; }
};
TEST_CASE("talk_26_1 external static & dynamic") {
print_area_dyn({Circle{5.0}, Rect{3.0, 4.0}});
print_area_static(as<Shape>(Circle{5.0}), as<Shape>(Rect{3.0, 4.0}));
}

No Visitors! Open Dispatch:
using namespace anyxx;
template <typename Self>
struct node : anyxx::deleteable<Self> {
using dispatch_table[[= v_table_data]] = anyxx::dispatch_table;
};
static_assert(is_open_dispatch_v_table<any<node>::v_table_t>);
struct Plus {
any<node> left_, right_;
};
struct Times {
any<node> left_, right_;
};
struct Integer {
int i;
};
// =============================================================================
// add behavior to existing classes, without changing them
//-----------------------------------------------------------------------------
// open dispatch evaluate, returns int, dispatched via an any<node>;
dispatch<int(virtual_<any<node>>)> evaluate;
auto __ = evaluate.define<Plus>([](auto const& expr) {
return evaluate(expr.left_) + evaluate(expr.right_);
});
auto __ = evaluate.define<Times>([](auto const& expr) {
return evaluate(expr.left_) * evaluate(expr.right_);
});
auto __ = evaluate.define<Integer>([](auto const& expr) { return expr.i; });
//
//-----------------------------------------------------------------------------
// render as Forth
dispatch<std::string(virtual_<any<node>>)> as_forth;
auto __ = as_forth.define<Plus>([](auto const& expr) {
return as_forth(expr.left_) + " " + as_forth(expr.right_) + " +";
});
auto __ = as_forth.define<Times>([](auto const& expr) {
return as_forth(expr.left_) + " " + as_forth(expr.right_) + " *";
});
auto __ = as_forth.define<Integer>(
[](auto const& expr) { return std::to_string(expr.i); });
//
auto expr = any<node>{std::in_place_type<Times>, Integer(2),
any<node>{std::in_place_type<Plus>,
any<node>{std::in_place_type<Integer>, 3},
any<node>{std::in_place_type<Integer>, 4}}};
auto v = evaluate(expr);
REQUIRE(v == 14);
std::stringstream out;
out << as_forth(expr) << " = " << as_lisp(expr) << " = " << evaluate(expr);
std::cout << out.str() << "\n";
REQUIRE(out.str() == "2 3 4 + * = (times 2 (plus 3 4)) = 14");

Open Multi Dispatch for Spaceship Operator
using namespace anyxx;
template <typename Self>
struct thing : copyable<Self> {
std::string registry() const;
using dispatch_table[[= anyxx::v_table_data]] = anyxx::dispatch_table;
};
dispatch<any<thing>(virtual_<any<thing>>, virtual_<any<thing>>)> collide;
any<thing> operator<=>(any<thing> const& l, any<thing> const& r) {
return collide(l, r);
}
any<thing> run(std::vector<any<thing>> things) {
return *std::ranges::fold_left_first(
things, [](any<thing> const& winner, any<thing> const& next) {
std::print("{} <=> {} -> ", winner.registry(), next.registry());
auto result = winner <=> next;
std::println("{}!", result.registry());
return result;
});
}
auto winner = run({Asteroid{"Chicxulub"}, Spaceship{"NCC-1701"},
Spaceship{"USGS Powell"}, DeathStar{"2"},
Asteroid{"Tunguska"}, DeathStar{"1"},
BlackHole{"Sagittarius A*"}, BlackHole{"Cygnus X-1"}});
CHECK(winner.registry() == "Black Hole Sagittarius A* | Cygnus X-1");

Vany Multi Dispatch: Fancy Expression Tree
template <typename Self>
struct evaluable : copyable<Self> {
std::string to_string() const;
using dispatch_table[[= v_table_data]] = anyxx::dispatch_table;
};
template <typename V>
requires std::formattable<V, char>
struct evaluable<V> {
static std::string to_string(V const& self) {
return std::format("{}", self);
}
};
using value = vany<evaluable, bool, int, double>;
using bool_int_double = vany_type_trait<value>::concrete_variant;
static_assert(std::is_same_v<bool_int_double, std::variant<bool, int, double>>);
template <typename Self>
struct node : anyxx::deleteable<Self> {
value evaluate() const;
using dispatch_table[[= v_table_data]] = anyxx::dispatch_table;
};
VANY_DISPACH_DECLARE(
, evaluate_plus,
(value(virtual_<any<evaluable>>, virtual_<any<evaluable>>)),
([]<no_any L, no_any R>(L&& l, R&& r) -> value
{ return std::forward<L>(l) + std::forward<R>(r);
}));
struct plus {
any<node> left_, right_;
value evaluate() const {
return evaluate_plus(left_.evaluate(), right_.evaluate());
}
};
template <>
struct evaluable<std::string> {
static std::string to_string(std::string const& self) { return self; };
};
auto __ = evaluate_plus.define<std::string, std::string>(
[](const auto& lhs, const auto& rhs) -> value { return lhs + rhs; });
auto __ = evaluate_plus.define<bool_int_double, std::string>(
[](const auto& lhs, const auto& rhs) -> value {
return as<evaluable>(lhs).to_string() + rhs;
});
auto __ = evaluate_plus.define<std::string, bool_int_double>(
[](const auto& lhs, const auto& rhs) -> value {
return lhs + as<evaluable>(rhs).to_string();
});
