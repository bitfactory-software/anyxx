273std::expected<ToAny, cast_error> borrow_as(FromProxy const& from,
274 FromVTable* from_v_table) {
275 using to = typename ToAny::proxy_t;
276 return query_v_table<ToAny>(from_v_table).transform([&](auto v_table) {
277 return ToAny{borrow_proxy_as<to>(from, v_table), v_table};

284template <is_any ToAny, is_any FromAny>
285 requires borrowable_from<typename ToAny::proxy_t,
286 typename FromAny::proxy_t,
287 typename FromAny::v_table_t>
288std::expected<ToAny, cast_error> borrow_as(FromAny const& from) {
289 if constexpr (is_any_derived_from<FromAny, ToAny>) {
290 return {ToAny{from}};
291 } else if constexpr (is_any_derived_from<ToAny, FromAny>) {
292 return *downcast_to<ToAny>(from);
293 } else {
294 return borrow_as<ToAny>(get_proxy(from), get_v_table(from));

331template <typename Concrete>
332mutable_void invoke_move_constructor([[maybe_unused]] mutable_void placement,
333 [[maybe_unused]] mutable_void from) {
334 static_assert(std::is_move_constructible_v<Concrete>);
335 auto typed_placement = static_cast<Concrete*>(placement);
336 auto typed_from = static_cast<Concrete*>(from);
337 [[maybe_unused]] auto constructed =
338 std::construct_at<Concrete>(typed_placement, std::move(*typed_from));
339 assert(placement == constructed);
340 std::destroy_at(typed_from);
341 return placement;
342}

347 [[maybe_unused]] const_void from) {

427template <typename Tag>
428struct key {
429 const char* label;
430 friend auto operator<=>(key, key) = default;

432template <typename T>
433struct is_key_impl : std::false_type {};
434template <typename T>
435struct is_key_impl<key<T>> : std::true_type {};
436template <typename T>
437concept is_key = is_key_impl<T>::value;

441template <typename Any, typename Key, typename... Args>
442 requires proxy_trait<typename Any::proxy_t>::is_owner
443class factory {
444 using constructor_t = std::function<Any(Args...)>;
445 std::map<Key, constructor_t> factory_map_;
446
447 auto register_impl(Key key, auto const& construct) {
448 factory_map_[key] = [construct](Args... args) -> Any {
449 return Any{std::in_place, construct(std::forward<Args>(args)...)};
450 };
451 }
452
453 auto construct_impl(Key key, Args... args) {
454 if (auto found = factory_map_.find(key); found != factory_map_.end())
455 return found->second(std::forward<Args>(args)...);
456 if constexpr (std::same_as<Key, std::string>) {
457 throw unkonwn_factory_key_error{key};
458 } else if constexpr (is_key<Key>) {
459 throw unkonwn_factory_key_error{key.label};
460 } else {
461 throw unkonwn_factory_key_error{std::to_string(key)};
462 }
463 };
464
465 public:
466 auto register_(Key const& key, auto const& construct) {
467 register_impl(key, construct);
468 return nullptr;
469 }
470 auto construct(auto key, Args&&... args) {
471 return construct_impl(key, std::forward<Args>(args)...);
472 }
473};

490template <typename InObject>
491struct members {
492 members() : table_(members_count<InObject>()) {}
493 using any_value_t = any<dynamic_copyable, nullable_val>;
494 std::vector<any_value_t> table_;
495 template <typename Member, typename Arg>
496 void set(Member member, Arg&& arg) {
497 using value_t = typename Member::value_t;
498 table_[member.index] =
499 any_value_t{std::in_place_type<value_t>, std::forward<Arg>(arg)};
500 }
501 template <typename Member>
502 typename Member::value_t const* get(Member member) const {
503 const auto& val = table_[member.index];
504 if (!val) return {};
505 return unchecked_unerase_cast<typename Member::value_t>(val);
506 }
507 template <typename Member>
508 typename Member::value_t* get(Member member) {
509 auto& val = table_[member.index];
510 if (!val) return {};
511 return unchecked_unerase_cast<typename Member::value_t>(val);
512 }
513 template <typename Member>
514 typename Member::value_t& operator[](Member member) {
515 if (auto val = get(member)) {
516 return *val;
517 }
518 using value_t = typename Member::value_t;
519 set(member, value_t());
520 return *get(member);
521 }
522};
