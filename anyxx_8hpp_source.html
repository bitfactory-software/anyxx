68template <is_any ToAny, is_any FromAny>
69 requires borrowable_from<typename ToAny::proxy_t,
70 typename FromAny::proxy_t,
71 typename FromAny::v_table_t>
72std::expected<ToAny, cast_error> borrow_as(FromAny const& from) {
73 if constexpr (is_any_derived_from<FromAny, ToAny>) {
74 return {ToAny{from}};
75 } else if constexpr (is_any_derived_from<ToAny, FromAny>) {
76 return *downcast_to<ToAny>(from);
77 } else {
78 return borrow_as<ToAny>(get_proxy(from), get_v_table(from));
79 }

85std::expected<ToAny, cast_error> clone_to(FromAny const& from) {
86 using vv_to_t = typename ToAny::proxy_t;
87 static_assert(is_proxy<vv_to_t>);
88 return query_v_table<ToAny>(get_v_table(from)).transform([&](auto v_table) {
89 return ToAny{clone_to<vv_to_t>(get_proxy(from), v_table), v_table};

95template <is_any FromAny>
96 requires std::same_as<typename FromAny::proxy_t, weak>
97auto lock(FromAny const& from_interface) {
98 using to_any_t = any<typename FromAny::trait_t, shared>;
99 static_assert(is_any<to_any_t>);
100 using return_t = std::optional<to_any_t>;
101 if (auto locked = get_proxy(from_interface).lock())
102 return return_t{to_any_t{std::move(locked), get_v_table(from_interface)}};
103 return return_t{};
104}

115template <typename Concrete>
116mutable_void invoke_move_constructor([[maybe_unused]] mutable_void placement,
117 [[maybe_unused]] mutable_void from) {
118 static_assert(std::is_move_constructible_v<Concrete>);
119 auto typed_placement = static_cast<Concrete*>(placement);
120 auto typed_from = static_cast<Concrete*>(from);
121 [[maybe_unused]] auto constructed =
122 std::construct_at<Concrete>(typed_placement, std::move(*typed_from));
123 assert(placement == constructed);
124 std::destroy_at(typed_from);
125 return placement;
126}

129 requires std::copy_constructible<Concrete>
130mutable_void invoke_copy_constructor([[maybe_unused]] mutable_void placement,
131 [[maybe_unused]] const_void from) {
132 return std::construct_at<Concrete>(static_cast<Concrete*>(placement),
133 *static_cast<Concrete const*>(from));
