1869 DynamicDispatch::dimension_count;

1871 template <typename Vany1, typename... Args>
1872 auto invoke1(Vany1&& vany, Args&&... args) const {
1873 return std::visit(
1874 [&]<typename TypedArg>([[maybe_unused]] TypedArg&& arg) {
1875 return overloads{
1876 StaticDispatch,
1877 [&]<is_any Any, typename... Vargs>(Any&& any, Vargs&&... vargs) {
1878 return dynamic_dispatch_(std::forward<Any>(any),
1879 std::forward<Vargs>(vargs)...);
1880 }}(std::forward<TypedArg>(arg), std::forward<Args>(args)...);
1881 },
1882 get_proxy_value(std::forward<Vany1>(vany)));
1883 }
1884
1885 template <is_any Vany1, is_any Vany2, typename... Args>
1886 auto invoke2(Vany1&& vany1, Vany2&& vany2, Args&&... args) const {
1887 using vany1_t = std::decay_t<Vany1>;
1888 using vany2_t = std::decay_t<Vany2>;
1889 using cv1_t = anyxx::vany_type_trait<vany1_t>::concrete_variant;
1890 using any_v1 = anyxx::vany_type_trait<vany1_t>::any_in_variant;
1891 using cv2_t = anyxx::vany_type_trait<vany2_t>::concrete_variant;
1892 using any_v2 = anyxx::vany_type_trait<vany2_t>::any_in_variant;
1893
1894 auto dispatch_combined = [&]<typename DA1, typename DA2>(DA1&& da1,
1895 DA2&& da2) {
1896 auto dyn_case1 = [&]<is_any A1, is_any A2, typename... VAs>(
1897 A1&& a1, A2&& a2, VAs&&... vas) {
1898 return dynamic_dispatch_(std::forward<A1>(a1), std::forward<A2>(a2),
1899 std::forward<VAs>(vas)...);
1900 };
1901 auto dyn_case2 = [&]<is_any A1, typename A2, typename... VAs>(
1902 A1&& a1, A2 a2, VAs&&... vas)
1903 requires std::constructible_from<cv2_t, A2>
1904 {
1905 return dynamic_dispatch_(std::forward<A1>(a1),
1906 any_v2{std::in_place, cv2_t{std::move(a2)}},
1907 std::forward<VAs>(vas)...);
1908 };
1909 auto dyn_case3 = [&]<typename A1, is_any A2, typename... VAs>(
1910 A1 a1, A2&& a2, VAs&&... vas)
1911 requires std::constructible_from<cv1_t, A1>
1912 {
1913 return dynamic_dispatch_(any_v1{std::in_place, cv1_t{std::move(a1)}},
1914 std::forward<A2>(a2),
1915 std::forward<VAs>(vas)...);
1916 };
1917 auto dyn_case4 = [&]<typename A1, typename A2, typename... VAs>(
1918 A1 a1, A2 a2, VAs&&... vas)
1919 requires(std::constructible_from<cv1_t, A1> &&
1920 std::constructible_from<cv2_t, A2>)
1921 {
1922 return dynamic_dispatch_(any_v1{std::in_place, cv1_t{std::move(a1)}},
1923 any_v2{std::in_place, cv2_t{std::move(a2)}},
1924 std::forward<VAs>(vas)...);
1925 };
1926 return overloads{StaticDispatch, dyn_case1, dyn_case2, dyn_case3,
1927 dyn_case4}(std::forward<DA1>(da1),
1928 std::forward<DA2>(da2),
1929 std::forward<Args>(args)...);
1930 };
1931
1932 return std::visit(dispatch_combined,
1933 get_proxy_value(std::forward<Vany1>(vany1)),
1934 get_proxy_value(std::forward<Vany2>(vany2)));
1935 }

1937 public:

1940 return dynamic_dispatch_.template define<Classes...>(f);
1941 }

1947 } else {
1948 if constexpr (dimension_count == 2) {
1949 return invoke2(std::forward<Args>(args)...);

1952 "dispatch_vany only supports one and two dimensions");
1953 }
