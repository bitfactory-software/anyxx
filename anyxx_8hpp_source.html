55template <typename Concrete>
56mutable_void invoke_move_constructor([[maybe_unused]] mutable_void placement,
57 [[maybe_unused]] mutable_void from) {
58 static_assert(std::is_move_constructible_v<Concrete>);
59 auto typed_placement = static_cast<Concrete*>(placement);
60 auto typed_from = static_cast<Concrete*>(from);
61 [[maybe_unused]] auto constructed =
62 std::construct_at<Concrete>(typed_placement, std::move(*typed_from));
63 assert(placement == constructed);
64 std::destroy_at(typed_from);
65 return placement;
66}

68template <typename Concrete>
69 requires std::copy_constructible<Concrete>
70mutable_void invoke_copy_constructor([[maybe_unused]] mutable_void placement,
71 [[maybe_unused]] const_void from) {
72 return std::construct_at<Concrete>(static_cast<Concrete*>(placement),
73 *static_cast<Concrete const*>(from));
74}
